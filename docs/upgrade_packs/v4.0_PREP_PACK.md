def now_iso() -> str:
    return datetime.datetime.now().astimezone().isoformat(timespec="seconds")

def ensure_dir(path: str) -> None:
    os.makedirs(path, exist_ok=True)

def read_text(path: str) -> str:
    with open(path, "r", encoding="utf-8") as f:
        return f.read()

def read_json(path: str) -> Dict[str, Any]:
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def write_json(path: str, obj: Dict[str, Any]) -> None:
    ensure_dir(os.path.dirname(path) or ".")
    with open(path, "w", encoding="utf-8") as f:
        json.dump(obj, f, ensure_ascii=False, indent=2)

def append_text(path: str, content: str) -> None:
    # append-only (creates file if missing)
    with open(path, "a", encoding="utf-8") as f:
        f.write(content)

def sha256_file(path: str) -> Optional[str]:
    if not os.path.exists(path):
        return None
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while True:
            b = f.read(1024 * 1024)
            if not b:
                break
            h.update(b)
    return "sha256:" + h.hexdigest()

def stable_short_hash(s: str) -> str:
    return hashlib.sha256(s.encode("utf-8")).hexdigest()[:8]

def parse_kv_md(md: str) -> Dict[str, str]:
    # Parse only KEY: VALUE lines where KEY is uppercase underscore.
    out: Dict[str, str] = {}
    for line in md.splitlines():
        if ":" in line:
            k, v = line.split(":", 1)
            k = k.strip()
            v = v.strip()
            if k and v and all(c.isupper() or c.isdigit() or c == "_" for c in k):
                out[k] = v
    return out

def safe_load_json(path: str) -> Optional[Dict[str, Any]]:
    try:
        if os.path.exists(path):
            return read_json(path)
    except Exception:
        return None
    return None

def classify_decision(state: Dict[str, Any], reports: Dict[str, Optional[Dict[str, Any]]], explicit: bool) -> Tuple[str, str, str]:
    """
    Returns (summary, action_taken, outcome).
    - summary: one-line
    - action_taken: allow/block/autofix/escalate
    - outcome: PASS/FAIL/WARNING
    """
    approval = str(state.get("approval_status", "PENDING"))
    mode = str(state.get("mode", "DRY_RUN"))
    phase = state.get("current_phase", None)

    # design policy
    dpol = reports.get("design_policy")
    dpol_status = None
    dpol_score = None
    dpol_crit = None
    if isinstance(dpol, dict):
        dpol_status = dpol.get("status")
        dpol_score = dpol.get("score_total")
        counts = dpol.get("counts", {})
        if isinstance(counts, dict):
            dpol_crit = counts.get("critical")

    # contrast
    cpol = reports.get("contrast")
    cpol_status = cpol.get("status") if isinstance(cpol, dict) else None

    # drift
    drift = reports.get("drift")
    drift_sev = drift.get("severity") if isinstance(drift, dict) else None

    # security
    sec = reports.get("security")
    sec_status = sec.get("status") if isinstance(sec, dict) else None
    sec_crit = sec.get("counts", {}).get("critical") if isinstance(sec, dict) else None

    # Determine decision meaning
    reasons: List[str] = []
    outcome = "PASS"
    action = "allow"

    # If approval not approved (explicit decisions)
    if explicit:
        if approval != "APPROVED":
            outcome = "FAIL"
            action = "block"
            reasons.append(f"approval_status={approval}")

    # Gate-like interpretation
    if dpol_status == "FAIL" or (isinstance(dpol_crit, int) and dpol_crit > 0):
        outcome = "FAIL"
        action = "block"
        reasons.append("design_policy=FAIL")

    if cpol_status == "FAIL":
        outcome = "FAIL"
        action = "block"
        reasons.append("contrast=FAIL")

    if drift_sev == "CRITICAL":
        outcome = "FAIL"
        action = "block"
        reasons.append("drift=CRITICAL")

    if sec_status == "FAIL" or (isinstance(sec_crit, int) and sec_crit > 0):
        outcome = "FAIL"
        action = "block"
        reasons.append("security=FAIL")

    # Warning tier
    if outcome == "PASS":
        if drift_sev == "WARNING":
            outcome = "WARNING"
            action = "allow"
            reasons.append("drift=WARNING")
        if dpol_score is not None and isinstance(dpol_score, (int, float)) and dpol_score < 75:
            outcome = "WARNING"
            reasons.append(f"design_score={dpol_score}")

    # Summary
    parts = []
    parts.append(f"mode={mode}")
    if phase is not None:
        parts.append(f"phase={phase}")
    if dpol_score is not None:
        parts.append(f"design_score={dpol_score}")
    if drift_sev:
        parts.append(f"drift={drift_sev}")
    if sec_status:
        parts.append(f"security={sec_status}")
    if reasons:
        parts.append("reasons=" + ",".join(reasons))

    summary = "Gate evaluation: " + " | ".join(parts)
    return summary, action, outcome

def build_evidence(reports: Dict[str, Optional[Dict[str, Any]]]) -> Dict[str, Any]:
    ev: Dict[str, Any] = {}
    dpol = reports.get("design_policy")
    if isinstance(dpol, dict):
        ev["design_policy"] = {
            "status": dpol.get("status"),
            "quality_tier": dpol.get("quality_tier"),
            "score_total": dpol.get("score_total"),
            "counts": dpol.get("counts"),
            "violations_count": dpol.get("violations_count"),
        }
    cpol = reports.get("contrast")
    if isinstance(cpol, dict):
        ev["contrast"] = {
            "status": cpol.get("status"),
            "required_ratio": cpol.get("required_ratio"),
            "violations": cpol.get("violations", [])[:10],
        }
    drift = reports.get("drift")
    if isinstance(drift, dict):
        ev["drift"] = {
            "severity": drift.get("severity"),
            "changed_keys_count": drift.get("changed_keys_count"),
            "changed_categories": drift.get("changed_categories"),
            "score_delta": drift.get("score_delta"),
        }
    sec = reports.get("security")
    if isinstance(sec, dict):
        ev["security"] = {
            "status": sec.get("status"),
            "counts": sec.get("counts"),
            "findings": sec.get("findings", [])[:10],
        }
    return ev

def load_inputs_hashes(root: str) -> Dict[str, Any]:
    out: Dict[str, Any] = {"files": {}}
    for rel in ANCHOR_FILES:
        p = os.path.join(root, rel)
        out["files"][rel] = sha256_file(p)
    for rel in REPORT_FILES:
        p = os.path.join(root, rel)
        if os.path.exists(p):
            out["files"][rel] = sha256_file(p)
    return out

def next_decision_id(root: str, day_dir: str) -> str:
    # DEC-YYYYMMDD-###-<short>
    d = datetime.datetime.now().strftime("%Y%m%d")
    # count existing
    n = 0
    if os.path.exists(day_dir):
        for fn in os.listdir(day_dir):
            if fn.startswith(f"DEC-{d}-") and fn.endswith(".json"):
                n += 1
    idx = n + 1
    # short hash: based on anchors hashes to avoid collisions
    hashes = load_inputs_hashes(root)
    seed = json.dumps(hashes, sort_keys=True)
    sh = stable_short_hash(seed)
    return f"DEC-{d}-{idx:03d}-{sh}"

def main():
    root = "."
    actor = "orchestrator"
    decision_type = "IMPLICIT"  # IMPLICIT/EXPLICIT
    log_md = DEFAULT_LOG_MD

    # args
    args = sys.argv[1:]
    if "--root" in args:
        root = args[args.index("--root")+1]
    if "--actor" in args:
        actor = args[args.index("--actor")+1]
    if "--type" in args:
        decision_type = args[args.index("--type")+1].upper()
    if "--log" in args:
        log_md = args[args.index("--log")+1]

    # state + reports
    state = safe_load_json(os.path.join(root, ".orchestrator/state.json")) or {}
    design_system_text = ""
    if os.path.exists(os.path.join(root, "DESIGN_SYSTEM.md")):
        design_system_text = read_text(os.path.join(root, "DESIGN_SYSTEM.md"))
    design_system_kv = parse_kv_md(design_system_text) if design_system_text else {}

    reports = {
        "design_policy": safe_load_json(os.path.join(root, ".orchestrator/design_policy_report.json")),
        "contrast": safe_load_json(os.path.join(root, ".orchestrator/contrast_report.json")),
        "drift": safe_load_json(os.path.join(root, ".orchestrator/design_drift.json")),
        "security": safe_load_json(os.path.join(root, ".orchestrator/security_report.json")),
    }

    explicit = (decision_type == "EXPLICIT")
    summary, action_taken, outcome = classify_decision(state, reports, explicit)
    evidence = build_evidence(reports)
    inputs_hashes = load_inputs_hashes(root)

    # id + output paths
    day = datetime.datetime.now().strftime("%Y-%m-%d")
    day_dir = os.path.join(root, ".orchestrator/decision_events", day)
    ensure_dir(day_dir)
    decision_id = next_decision_id(root, day_dir)
    out_json = os.path.join(day_dir, f"{decision_id}.json")

    event = {
        "decision_id": decision_id,
        "generated_at": now_iso(),
        "actor": actor,
        "type": decision_type,
        "summary": summary,
        "action_taken": action_taken,   # allow/block/autofix/escalate
        "outcome": outcome,             # PASS/FAIL/WARNING
        "context": {
            "DESIGN_MODE": design_system_kv.get("DESIGN_MODE"),
            "ACCESSIBILITY_LEVEL": design_system_kv.get("ACCESSIBILITY_LEVEL"),
            "PRIMARY_CONTEXT": design_system_kv.get("PRIMARY_CONTEXT"),
            "approval_status": state.get("approval_status"),
            "mode": state.get("mode"),
            "current_phase": state.get("current_phase"),
        },
        "inputs": inputs_hashes,
        "evidence": evidence,
    }

    write_json(out_json, event)

    # append MD
    md_lines = []
    md_lines.append("")
    md_lines.append("## " + decision_id)
    md_lines.append(f"- GeneratedAt: {event['generated_at']}")
    md_lines.append(f"- Actor: {actor}")
    md_lines.append(f"- Type: {decision_type}")
    md_lines.append(f"- Outcome: {outcome}")
    md_lines.append(f"- Action: {action_taken}")
    md_lines.append(f"- Summary: {summary}")
    md_lines.append("- Evidence:")
    if "design_policy" in evidence:
        dp = evidence["design_policy"]
        md_lines.append(f"  - design_policy: status={dp.get('status')} score={dp.get('score_total')} counts={dp.get('counts')}")
    if "contrast" in evidence:
        cp = evidence["contrast"]
        md_lines.append(f"  - contrast: status={cp.get('status')} required_ratio={cp.get('required_ratio')}")
    if "drift" in evidence:
        dr = evidence["drift"]
        md_lines.append(f"  - drift: severity={dr.get('severity')} changed_keys={dr.get('changed_keys_count')} score_delta={dr.get('score_delta')}")
    if "security" in evidence:
        sr = evidence["security"]
        md_lines.append(f"  - security: status={sr.get('status')} counts={sr.get('counts')}")
    md_lines.append(f"- EventFile: {out_json}")
    md_lines.append("")

    append_text(os.path.join(root, log_md), "\n".join(md_lines))

    print("OK: decision event written:", out_json)
    # If outcome FAIL, return 1 so CI can use it as a gate if desired
    if outcome == "FAIL":
        sys.exit(1)
    sys.exit(0)

if __name__ == "__main__":
    try:
        main()
    except Exception as e:
        print("ERROR:", str(e))
        sys.exit(2)

# 🔐 SECURITY_POLICY_RULES.md (v4.0 Draft Template)
# 목적: 보안/컴플라이언스 정책을 OPA 스타일(when/must/evidence/severity)로 선언하고, 스크립트가 이를 검사하여 게이트로 사용한다.

## 0) Policy Identity
- POLICY_DOMAIN: SECURITY
- POLICY_VERSION: 4.0-draft
- OWNER: Security Owner
- ENFORCEMENT: STRICT (STRICT | WARN_ONLY)
- DEFAULT_SEVERITY: MAJOR

## 1) Evidence Requirements (릴리즈/PR 공통)
다음 Evidence가 없으면 정책 검사를 FAIL로 간주한다.
- .orchestrator/security_report.json
- (선택) .orchestrator/sbom.json
- (선택) .orchestrator/license_report.json
- (선택) .orchestrator/secret_scan_report.json

## 2) Severity Model
- CRITICAL: 즉시 차단 (merge/release 불가)
- MAJOR: 기본 차단(STRICT), WARN_ONLY 모드에서는 경고
- MINOR: 경고 + 개선 권고

## 3) Policy Rules (OPA-style)

### SEC-001: No Secrets in Repo
- WHEN:
  - any file contains patterns matching secrets (API keys, tokens, private keys)
- MUST_NOT:
  - secrets exist in tracked files
- EVIDENCE:
  - secret_scan_report.json (tool: gitleaks) includes "findings" == 0
- SEVERITY: CRITICAL
- REMEDIATION:
  - remove secrets, rotate keys, add to .gitignore, add secret scanning allowlist only with approval

### SEC-002: Dependency Vulnerabilities (Known Exploits)
- WHEN:
  - dependencies are present (package-lock/yarn.lock/pnpm-lock/pubspec.lock)
- MUST:
  - no CRITICAL vulnerabilities (and no HIGH in STRICT mode)
- EVIDENCE:
  - security_report.json includes vuln_summary.critical == 0
- SEVERITY: CRITICAL
- REMEDIATION:
  - upgrade packages, patch versions, replace dependency

### SEC-003: SAST Findings Threshold
- WHEN:
  - source code exists under apps/ or shared/
- MUST:
  - sast.critical == 0
  - sast.high <= 3 (configurable)
- EVIDENCE:
  - security_report.json includes sast_summary
- SEVERITY: MAJOR
- REMEDIATION:
  - fix injection, unsafe eval, insecure deserialization, SSRF, auth bypass

### SEC-004: Approved Cryptography Only
- WHEN:
  - code uses crypto primitives
- MUST_NOT:
  - custom crypto implementation
  - weak algorithms (MD5, SHA1 for security, ECB mode)
- EVIDENCE:
  - sast report flags do not include weak-crypto
- SEVERITY: MAJOR
- REMEDIATION:
  - use well-reviewed libs, modern algorithms, vetted modes

### SEC-005: Transport Security
- WHEN:
  - backend or API calls exist
- MUST:
  - https only in production configs
  - no plaintext credentials in URLs
- EVIDENCE:
  - config scan results show no http endpoints in prod
- SEVERITY: MAJOR
- REMEDIATION:
  - enforce HTTPS, use secure headers, move secrets to env/secret manager

### SEC-006: Build Provenance (SBOM)
- WHEN:
  - release pipeline runs
- MUST:
  - produce SBOM artifact
- EVIDENCE:
  - sbom.json exists with components list
- SEVERITY: MINOR (CRITICAL for regulated projects)
- REMEDIATION:
  - enable SBOM generator (cyclonedx/syft)

### SEC-007: License Compliance (Allowed Licenses)
- WHEN:
  - dependencies exist
- MUST:
  - no forbidden licenses (e.g., GPL-3.0 in proprietary context)
- EVIDENCE:
  - license_report.json forbidden_count == 0
- SEVERITY: MAJOR
- REMEDIATION:
  - replace dependency, obtain commercial license, isolate with legal approval

## 4) Policy Parameters (프로젝트별 재정의 가능)
- VULN_BLOCK_LEVEL: CRITICAL (CRITICAL | HIGH | MEDIUM)
- SAST_HIGH_MAX: 3
- LICENSE_FORBIDDEN: [GPL-3.0, AGPL-3.0]
- SECRET_SCAN_TOOL: gitleaks

## 5) Output Contract (for checkers)
보안 검사 스크립트는 아래 형식으로 결과를 저장해야 한다.
- Path: .orchestrator/security_report.json

    "secret_summary": {"findings": 0},
    "findings": [
      {"rule_id":"SEC-001","severity":"CRITICAL","path":"...","evidence":"...","remediation":"..."}
    ]
  }
